cmake_minimum_required(VERSION 3.21)

find_package(PkgConfig REQUIRED)
pkg_check_modules(Criterion REQUIRED IMPORTED_TARGET criterion)

foreach(test test_builder test_channels test_gold_timestamp test_null_guards test_thread_local
             test_value_ranges)
    add_executable(${test} ${test}.c)
    target_link_libraries(${test} PRIVATE GeoTrace::C PkgConfig::Criterion geotrace_sanitizers)

    if(MSVC)
        target_compile_options(${test} PRIVATE /W4 /WX)
    else()
        target_compile_options(${test} PRIVATE
            -Wall -Wextra -Wpedantic
            -Wshadow -Wstrict-prototypes -Wmissing-prototypes
            -Wcast-qual -Wformat=2 -Wmissing-declarations -Wconversion
            -Werror
        )
    endif()

    # Embed the library directory in the binary's RPATH so `ctest` finds the
    # shared library without needing LD_LIBRARY_PATH / DYLD_LIBRARY_PATH.
    set_target_properties(${test} PROPERTIES
        BUILD_RPATH "$<TARGET_FILE_DIR:GeoTrace::C>"
    )

    add_test(NAME ${test} COMMAND ${test})
    set_tests_properties(${test} PROPERTIES
        ENVIRONMENT "${GEOTRACE_SANITIZE_TEST_ENVIRONMENT}"
    )
endforeach()

# Round-trip test uses the minimal fixture generated by `gen_fixture`.
set(GTD_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/minimal.gtd")
if(EXISTS "${GTD_FIXTURE}")
    target_compile_definitions(test_builder PRIVATE
        GTD_FIXTURE_PATH="${GTD_FIXTURE}"
    )
endif()

# The value-range test reads the out-of-range fixture generated by `gen_fixture`.
set(GTD_OUT_OF_RANGE_FIXTURE "${CMAKE_CURRENT_SOURCE_DIR}/fixtures/out_of_range_values.gtd")
if(EXISTS "${GTD_OUT_OF_RANGE_FIXTURE}")
    target_compile_definitions(test_value_ranges PRIVATE
        GTD_OUT_OF_RANGE_FIXTURE_PATH="${GTD_OUT_OF_RANGE_FIXTURE}"
    )
endif()
